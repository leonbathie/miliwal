    function nombreEnFulfulde(n) {
        n = parseInt(n);
        if (n < 0 || n > 59 || isNaN(n)) return n.toString();
        if (n < 10) return unitesFF[n];
        if (dizainesFF[n]) return dizainesFF[n];
        let d = Math.floor(n / 10) * 10;
        let u = n % 10;
        return dizainesFF[d] + " e " + unitesFF[u];
    }

    function tempsEnFulfulde(hours, minutes, seconds) {
        let hStr = nombreEnFulfulde(hours);
        let mStr = nombreEnFulfulde(minutes);
        let sStr = nombreEnFulfulde(seconds);
        let mWord = minutes <= 1 ? "hojom" : "hojomaaji";
        let sWord = seconds <= 1 ? "leƳƳande" : "leƳƳanɗe";
        return `Yamnde ${hStr}, ${mWord} ${mStr}, ${sWord} ${sStr}`;
    }

    function temperatureEnFulfulde(temp) {
        return `Tolno ngulɓuuɓdi ko ${Math.round(temp)}°`;
    }

    function formatRA(radians) {
        let hours = (radians * 180 / Math.PI) / 15;
        if(hours < 0) hours += 24;
        let h = Math.floor(hours);
        let m = Math.floor((hours - h) * 60);
        let s = Math.round((((hours - h) * 60) - m) * 60);
        return `${h}h ${m}m ${s}s`;
    }

    function formatDec(radians) {
        let degrees = radians * 180 / Math.PI;
        let sign = degrees >= 0 ? "+" : "-";
        degrees = Math.abs(degrees);
        let d = Math.floor(degrees);
        let m = Math.floor((degrees - d) * 60);
        let s = Math.round((((degrees - d) * 60) - m) * 60);
        return `${sign}${d}° ${m}' ${s}"`;
    }

    function buildCalendar(year) {
        const grid = document.getElementById('calendar-grid');
        grid.innerHTML = '';
        const today = new Date();
        
        for (let m = 0; m < 12; m++) {
            const card = document.createElement('div');
            card.className = 'month-card';
            
            let mainMonth = i18n[currentLang].months[m];
            let subMonth = (currentLang !== 'ff') ? i18n['ff'].months[m] : i18n['fr'].months[m];

            let seasonIcon = '';
            if ([11, 0, 1].includes(m)) seasonIcon = '❄️';
            else if ([2, 3, 4].includes(m)) seasonIcon = '☀️';
            else if ([5, 6, 7, 8].includes(m)) seasonIcon = '🌧️';
            else if ([9, 10].includes(m)) seasonIcon = '🌾';

            let html = `
                <div class="month-header">
                    <div class="month-season-icon">${seasonIcon}</div>
                    <h3>${m+1}. ${mainMonth}</h3>
                    <span>${subMonth} ${year}</span>
                </div>
                <div class="days-grid">
            `;
            
            i18n[currentLang].daysAbr.forEach(jour => {
                html += `<div class="day-label">${jour}</div>`;
            });

            const firstDay = new Date(year, m, 1).getDay();
            const emptyDays = firstDay === 0 ? 6 : firstDay - 1;
            
            for (let i = 0; i < emptyDays; i++) {
                html += `<div class="day-cell empty"></div>`;
            }

            const daysInMonth = new Date(year, m + 1, 0).getDate();
            for (let d = 1; d <= daysInMonth; d++) {
                let isToday = (d === today.getDate() && m === today.getMonth() && year === today.getFullYear()) ? 'today' : '';
                html += `<div class="day-cell ${isToday}">${d}</div>`;
            }

            html += `</div>`;
            card.innerHTML = html;
            grid.appendChild(card);
        }
    }

    function startClock() {
        function updateClock() {
            const now = new Date();
            const h = now.getHours(); const m = now.getMinutes(); const s = now.getSeconds();
            
            setHTML('clock-num', `${h.toString().padStart(2,'0')}:${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`);
            setText('clock-text', tempsEnFulfulde(h, m, s));

            let dayIndex = now.getDay() === 0 ? 6 : now.getDay() - 1;
            let dayName = i18n[currentLang].days[dayIndex];
            let monthName = i18n[currentLang].months[now.getMonth()];
            setText('clock-date', `${dayName}, ${now.getDate()} ${monthName} ${now.getFullYear()}`);
            setText('dash-time', `${h.toString().padStart(2,'0')}:${m.toString().padStart(2,'0')}`);
            
            let momentKey = 'moment_night';
            if (h >= 5 && h < 12) momentKey = 'moment_morning';
            else if (h >= 12 && h < 14) momentKey = 'moment_day';
            else if (h >= 14 && h < 18) momentKey = 'moment_afternoon';
            else if (h >= 18 && h < 20) momentKey = 'moment_evening';
            
            setText('clock-moment', i18n[currentLang][momentKey]);

            // Données temporelles scientifiques (JD et LST)
            let jd = (now.getTime() / 86400000) + 2440587.5;
            const d = jd - 2451545.0;
            let gmst = (280.46061837 + 360.98564736629 * d) % 360;
            if (gmst < 0) gmst += 360;
            let lst = (gmst + currentLon) % 360;
            if (lst < 0) lst += 360;
            let lstHours = lst / 15;
            let lstH = Math.floor(lstHours);
            let lstM = Math.floor((lstHours - lstH) * 60);
            
            setHTML('sci-time-data', `
                <span data-i18n="lbl_jd">${i18n[currentLang].lbl_jd}</span>: ${jd.toFixed(2)} | 
                <span data-i18n="lbl_lst">${i18n[currentLang].lbl_lst}</span>: ${lstH.toString().padStart(2,'0')}h${lstM.toString().padStart(2,'0')}
            `);
        }
        updateClock();
        setInterval(updateClock, 1000);
    }

    async function loadWeather(lat, lon) {
        // Ajout de wind_gusts_10m et daily uv_index_max

            const response = await fetch(url);
            const data = await response.json();

            renderWeather(data);
            renderChart(data);
            renderDailyForecast(data);
            if (typeof SunCalc !== 'undefined') {

            }
        } catch (err) {
            console.error("API Error", err);
        }
    }

    const PI = Math.PI, rad = PI/180, deg = 180/PI;

    function formatTimeOnly(date) {
        if(!date || isNaN(date.getTime())) return "--:--";
        return `${date.getHours().toString().padStart(2,'0')}:${date.getMinutes().toString().padStart(2,'0')}`;
    }

    function calculateRealAstrology(date, lat, lon) {
        const jd = date.getTime() / 86400000 + 2440587.5;
        const d = jd - 2451545.0;

        let L = (280.460 + 0.9856474 * d) % 360;
        let g = (357.528 + 0.9856003 * d) % 360;
        let lambda = (L + 1.915 * Math.sin(g * rad) + 0.020 * Math.sin(2 * g * rad)) % 360;
        if (lambda < 0) lambda += 360;
        const sunSignIdx = Math.floor(lambda / 30);

        return { sunIdx: sunSignIdx };
    }

        const sunCompass = document.getElementById('compass-sun');

        if(sunCompass) sunCompass.style.transform = `rotate(${sunAzDeg}deg)`;
        if(moonCompass) moonCompass.style.transform = `rotate(${moonAzDeg}deg)`;

    }

    function drawArc(canvasId, value, color) {
        const canvas = document.getElementById(canvasId);
        if(!canvas) return;
        const ctx = canvas.getContext('2d');
        const w = canvas.width; const h = canvas.height;
        const cx = w/2; const cy = h - 20; const r = h - 40;

        ctx.clearRect(0,0,w,h);
        ctx.beginPath(); ctx.arc(cx, cy, r, PI, 0);
        ctx.lineWidth = 10; ctx.strokeStyle = "rgba(255,255,255,0.1)"; ctx.stroke();

        let angleRatio = value > 0 ? Math.min(value / 90, 1) : 0;
        let endAngle = PI + (angleRatio * PI);

        if (value > 0) {
            ctx.beginPath(); ctx.arc(cx, cy, r, PI, endAngle);
            ctx.lineWidth = 10; ctx.strokeStyle = color; ctx.stroke();
            let ix = cx + r * Math.cos(endAngle); let iy = cy + r * Math.sin(endAngle);
            ctx.beginPath(); ctx.arc(ix, iy, 8, 0, 2*PI);
            ctx.fillStyle = color; ctx.fill();
        }
        ctx.beginPath(); ctx.moveTo(10, cy); ctx.lineTo(w-10, cy);
        ctx.lineWidth = 2; ctx.strokeStyle = "var(--plum-light)"; ctx.stroke();
    }
